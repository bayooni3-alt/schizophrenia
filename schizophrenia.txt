# Function to create transition matrix between two time points
create_transition_matrix <- function(data, time1, time2) {
  data %>%
    filter(!is.na(.[[time1]]), !is.na(.[[time2]])) %>%
    with(table(.[[time1]], .[[time2]]))
}

# Create transition matrix from month 0 to month 2
trans_0_2 <- create_transition_matrix(state_data, "month_0", "month_2")
print("Transition from Month 0 to Month 2:")
print(trans_0_2)
print(prop.table(trans_0_2, margin = 1)) # Row percentages (transition probabilities)

# Prepare data for log-linear analysis
# Create a dataset of all transitions
transition_data <- df %>%
  arrange(subject, month) %>%
  group_by(subject) %>%
  mutate(next_disorder = lead(disorder),
         next_month = lead(month)) %>%
  filter(!is.na(next_disorder)) %>%
  ungroup() %>%
  dplyr::select(subject, onset, month, disorder, next_month, next_disorder) %>%  # Use dplyr::select explicitly
  filter(disorder != "" & next_disorder != "") # Remove empty factors if needed

# Ensure factors are properly set
transition_data$disorder <- factor(transition_data$disorder)
transition_data$next_disorder <- factor(transition_data$next_disorder)
transition_data$onset <- factor(transition_data$onset)

# Create a contingency table for loglm
transition_table <- with(transition_data, 
                         table(disorder, next_disorder, onset))

# Now fit log-linear model correctly
loglin_model <- loglm(~ disorder + next_disorder + onset + disorder:next_disorder, 
                      data = transition_table)

summary(loglin_model)
